% -----------------------------------------------------------------------------
% Usage notes
% -----------------------------------------------------------------------------
% Paragraph spacing convention: this preamble does NOT use \usepackage{parskip}.
% Paragraphs are separated by blank lines in the source, which by default give
% LaTeX's standard paragraph indent and no extra vertical space. To match the
% FMM institutional style — visible blank line between paragraphs, no first-line
% indent — add \bigskip between paragraphs manually in the body:
%
%     First paragraph ends here.
%
%     \bigskip
%
%     Second paragraph begins here.
%
% This is a deliberate choice: per-document control is more predictable than a
% global parskip setting, which interacts poorly with titles, lists, and tables.
%
% In-text citations (\citep, \citet) render in BLACK (citecolor=black). URLs
% and internal cross-references remain PrimaryBlue. If a document type calls
% for coloured citations (e.g. an academic paper), override \hypersetup in the
% document preamble.
% -----------------------------------------------------------------------------

% Title formatting
% NOTE (April 2026): Title is fixed at 14pt to match the Word preamble ceiling.
% Previously used \LARGE (~17pt). Changed for cross-format consistency.
% The 14pt max is a hard design decision — Word titles looked too large at
% bigger sizes. LaTeX \section uses \Large (~14pt), so title and H1 are
% currently the same size; a future revision could differentiate them
% (e.g., \section at 13pt) to match the Word hierarchy (14pt title / 13pt H1).
\usepackage{titling}
\pretitle{\begin{center}\fontsize{14}{16.8}\selectfont\bfseries\color{PrimaryBlue}}
\posttitle{\end{center}\vskip 0.5em}
\preauthor{\begin{center}\large\color{SecondaryBlue}}
\postauthor{\end{center}}
\predate{\begin{center}\large\color{SecondaryBlue}}
\postdate{\end{center}}

% Customize itemize and enumerate
\usepackage{enumitem}
\setlist[itemize]{label=\textcolor{PrimaryBlue}{\textbullet}}
\setlist[enumerate]{label=\textcolor{PrimaryBlue}{\arabic*.}}

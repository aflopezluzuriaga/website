\documentclass[aspectratio=169, 10pt]{beamer}

% --- Theme — Boadilla with custom colors --------------------------------------
\usetheme{Boadilla}
\usepackage{lmodern}
\usepackage[utf8]{inputenc}
\usepackage[table]{xcolor}
\usepackage{booktabs}
\usepackage{makecell}

% --- Beamer theme colors -----------------------------------------------------
\setbeamercolor{palette primary}{bg=PrimaryBlue,fg=white}
\setbeamercolor{palette secondary}{bg=SecondaryBlue,fg=black}
\setbeamercolor{palette tertiary}{bg=LightBlue,fg=black}
\setbeamercolor{palette quaternary}{bg=PrimaryBlue,fg=white}
\setbeamercolor{author in head/foot}{bg=PrimaryBlue,fg=white}
\setbeamercolor{title in head/foot}{bg=SecondaryBlue,fg=black}
\setbeamercolor{date in head/foot}{bg=LightBlue,fg=black}
\setbeamercolor{structure}{fg=PrimaryBlue}
\setbeamercolor{frametitle}{bg=PrimaryBlue,fg=white}
\setbeamercolor{normal text}{fg=black}
\setbeamercolor{itemize item}{fg=SecondaryBlue}
\setbeamercolor{itemize subitem}{fg=SecondaryBlue}
\setbeamercolor{title}{fg=PrimaryBlue}
\setbeamercolor{author}{fg=black}
\setbeamercolor{institute}{fg=black}
\setbeamercolor{date}{fg=black}

% --- Packages ----------------------------------------------------------------
\usepackage{tcolorbox}
\tcbuselibrary{skins,breakable}
\usepackage{graphicx}

\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  urlcolor=PrimaryBlue, linkcolor=PrimaryBlue, citecolor=PrimaryBlue,
  pdfborder={0 0 1}
}
\usepackage{amsmath,amssymb}

% --- IDB logo on non-title slides (bottom-left, 0.35in) -----------------------
\addtobeamertemplate{frametitle}{}{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=south west,inner sep=0pt] at ($(current page.south west)+(0.3cm,0.3cm)$) {
      \includegraphics[height=0.35in]{IDBLogo.png}
    };
  \end{tikzpicture}
}

% --- Content spacing ---------------------------------------------------------
\setlength{\parskip}{0.8\baselineskip}
\setbeamerfont{itemize/enumerate body}{size=\normalsize}
\setbeamerfont{itemize/enumerate subbody}{size=\normalsize}
\setlength{\itemsep}{0.8em}
\let\olditemize\itemize
\renewcommand{\itemize}{\olditemize\setlength{\itemsep}{0.8em}\vspace{0.3em}}
\setbeamersize{text margin left=0.25in, text margin right=0.25in}

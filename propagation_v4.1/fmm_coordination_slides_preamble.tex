% \renewcommand{\familydefault}{\sfdefault}; Beamer's default font theme is
% already sans, so no font directives are needed.
\usepackage[utf8]{inputenc}

% --- Colors — mirror andreas_palette.md --------------------------------------

% --- Packages ----------------------------------------------------------------
\usepackage{tcolorbox}
\tcbuselibrary{skins,breakable}
\usepackage{graphicx}
\usepackage{tikz}
\usepackage{makecell}
\usetikzlibrary{shapes.geometric, arrows.meta, positioning, calc}
\usepackage{hyperref}
\hypersetup{
  colorlinks=true,
  urlcolor=PrimaryBlue,
  linkcolor=PrimaryBlue,
  citecolor=PrimaryBlue,
  pdfborder={0 0 1}
}

% --- Navigation and footline -------------------------------------------------
\setbeamertemplate{navigation symbols}{}
\setbeamertemplate{footline}{
  \ifnum\insertframenumber>1
    \begin{beamercolorbox}[wd=\paperwidth,ht=0.5in,dp=0ex,center]{footline}%
      \raisebox{0.2in}{\usebeamerfont{footline}\textcolor{PrimaryBlue}{\insertframenumber}}
    \end{beamercolorbox}%
  \fi
}

% --- Backgrounds — change BackgroundBodyWhite.png to BackgroundBody.png ------
%     here if Andrea chose the light-blue body background.
\defbeamertemplate{background canvas}{title}{
  \includegraphics[width=\paperwidth,height=\paperheight]{BackgroundTitle.png}
}
\defbeamertemplate{background canvas}{body}{
  \includegraphics[width=\paperwidth,height=\paperheight]{BackgroundBodyWhite.png}
}
\setbeamertemplate{background canvas}[body]

% --- Color scheme ------------------------------------------------------------
\setbeamercolor{frametitle}{fg=PrimaryBlue}
\setbeamercolor{structure}{fg=SecondaryBlue}
\setbeamercolor{normal text}{fg=black}
\setbeamercolor{item}{fg=SecondaryBlue}
\setbeamercolor{subitem}{fg=SecondaryBlue}
\setbeamerfont{footline}{size=\small}

% --- Spacing -----------------------------------------------------------------
\setlength{\parskip}{0.8\baselineskip}
\setbeamerfont{itemize/enumerate body}{size=\normalsize}
\setbeamerfont{itemize/enumerate subbody}{size=\normalsize}
\setlength{\itemsep}{0.8em}
\let\olditemize\itemize
\renewcommand{\itemize}{\olditemize\setlength{\itemsep}{0.8em}\vspace{0.3em}}

% --- Title page — white text on dark background -----------------------------
\setbeamerfont{title}{series=\bfseries}
\setbeamercolor{title}{fg=white}
\setbeamercolor{date}{fg=white}
